# Contatos: plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Criar, editar, excluir e listar contatos no Inbox, com sincronização na conta do WhatsApp.

**Architecture:** O servidor grava primeiro no WhatsApp (`addOrEditContact`/`removeContact`) e só depois no banco, numa regra pura em `server/contacts.ts`. A interface ganha um formulário único, um painel que desliza sobre a `list-pane` e uma lista de contatos compartilhada.

**Tech Stack:** Node 24, TypeScript com `node --test`, SQLite (`Store`), zod, React 19, CSS com tokens `--bt-*`, Baileys 7.0.0-rc14.

**Spec:** `docs/superpowers/specs/2026-11-06-contatos-design.md`

```ts
import type { Contact, Store } from "./db.ts";

type SaveDeps = {
  /** JID do número no WhatsApp; null se não tiver conta. */
  resolve: (digits: string) => Promise<string | null>;
  sync: (jid: string, fullName: string, firstName: string) => Promise<void>;
  store: Pick<Store, "setNames">;
};

/** WhatsApp primeiro: se a conta recusar, nada muda aqui e o Inbox não mostra nome que a conta não tem. */
export async function saveContact(deps: SaveDeps, input: { digits: string; firstName: string; lastName?: string }): Promise<Contact> {
  const firstName = input.firstName.trim();
  const fullName = [firstName, input.lastName?.trim()].filter(Boolean).join(" ");
  const jid = await deps.resolve(input.digits);
  if (!jid) throw new Error("Este número não tem WhatsApp.");
  await deps.sync(jid, fullName, firstName);
  deps.store.setNames(jid, { saved: fullName });
  return { jid, name: fullName, phone: jid.split("@")[0] };
}
```

```ts
import { useEffect, useSyncExternalStore } from "react";
import { api, type Contact } from "./api.ts";
import { normalize } from "./format.ts";

/** Nome é a primeira palavra; o resto é sobrenome. O WhatsApp guarda o nome completo. */
export function splitName(name: string): { firstName: string; lastName: string } {
  const [firstName = "", ...rest] = name.trim().split(/\s+/);
  return { firstName, lastName: rest.join(" ") };
}

export type ContactSection = { letter: string; contacts: Contact[] };

/** Filtra por nome ou dígitos e agrupa por inicial; o que não começa com letra vai para "#". */
export function contactSections(list: Contact[], query: string): ContactSection[] {
  const q = normalize(query.trim());
  const digits = query.replace(/\D/g, "");
  const hits = list.filter((c) => !q || normalize(c.name).includes(q) || (digits.length > 0 && c.phone.includes(digits)));
  const sections = new Map<string, Contact[]>();
  for (const c of hits) {
    const first = normalize(c.name).charAt(0).toUpperCase();
    const letter = /[A-Z]/.test(first) ? first : "#";
    sections.set(letter, [...(sections.get(letter) ?? []), c]);
  }
  return [...sections].map(([letter, contacts]) => ({ letter, contacts })).sort((a, b) => a.letter.localeCompare(b.letter, "pt-BR"));
}

type State = { contacts: Contact[] | null; error: string | null };
let state: State = { contacts: null, error: null };
const listeners = new Set<() => void>();
const set = (next: State) => {
  state = next;
  listeners.forEach((l) => l());
};
const byName = (a: Contact, b: Contact) => a.name.localeCompare(b.name, "pt-BR", { sensitivity: "base" });

export async function refreshContacts(): Promise<Contact[]> {
  try {
    const contacts = await api.contacts();
    set({ contacts, error: null });
    return contacts;
  } catch (e) {
    set({ ...state, error: (e as Error).message });
    throw e;
  }
}

export function markSaved(contact: Contact) {
  set({ ...state, contacts: [...(state.contacts ?? []).filter((c) => c.jid !== contact.jid), contact].sort(byName) });
}

export function markDeleted(jid: string) {
  set({ ...state, contacts: (state.contacts ?? []).filter((c) => c.jid !== jid) });
}

/** Agenda compartilhada por painel, perfil e cartões; carrega na primeira tela que precisar. */
export function useContacts(): State {
  const value = useSyncExternalStore((l) => (listeners.add(l), () => void listeners.delete(l)), () => state);
  useEffect(() => {
    if (state.contacts === null) void refreshContacts().catch(() => undefined);
  }, []);
  return value;
}
```
